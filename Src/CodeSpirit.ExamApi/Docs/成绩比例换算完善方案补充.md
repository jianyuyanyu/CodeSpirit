```csharp
namespace CodeSpirit.ExamApi.Models;

/// <summary>
/// 成绩显示信息
/// </summary>
public class ScoreDisplayInfo
{
    /// <summary>
    /// 原始成绩
    /// </summary>
    public double OriginalScore { get; set; }
    
    /// <summary>
    /// 原始满分
    /// </summary>
    public int OriginalFullScore { get; set; }
    
    /// <summary>
    /// 换算后成绩
    /// </summary>
    public double? ConvertedScore { get; set; }
    
    /// <summary>
    /// 换算后满分
    /// </summary>
    public int? ConvertedFullScore { get; set; }
    
    /// <summary>
    /// 是否启用了成绩换算
    /// </summary>
    public bool IsScoreConverted { get; set; }
    
    /// <summary>
    /// 换算比例
    /// </summary>
    public decimal? ConversionRatio { get; set; }
    
    /// <summary>
    /// 换算描述
    /// </summary>
    public string ConversionDescription { get; set; } = string.Empty;
    
    /// <summary>
    /// 显示成绩（优先显示换算后成绩）
    /// </summary>
    public double DisplayScore => ConvertedScore ?? OriginalScore;
    
    /// <summary>
    /// 显示满分（优先显示换算后满分）
    /// </summary>
    public int DisplayFullScore => ConvertedFullScore ?? OriginalFullScore;
    
    /// <summary>
    /// 得分率
    /// </summary>
    public double ScoreRate => DisplayFullScore > 0 ? (DisplayScore / DisplayFullScore) * 100 : 0;
    
    /// <summary>
    /// 成绩等级文本
    /// </summary>
    public string GradeText
    {
        get
        {
            var rate = ScoreRate;
            return rate switch
            {
                >= 90 => "优秀",
                >= 80 => "良好", 
                >= 70 => "中等",
                >= 60 => "及格",
                _ => "不及格"
            };
        }
    }
    
    /// <summary>
    /// 是否通过
    /// </summary>
    public bool IsPassed { get; set; }
    
    /// <summary>
    /// 成绩状态CSS类
    /// </summary>
    public string StatusCssClass => IsPassed ? "passed" : "failed";
    
    /// <summary>
    /// 换算标识CSS类
    /// </summary>
    public string ConversionCssClass => IsScoreConverted ? "converted" : "original";
}
```

### 1.3 成绩显示服务实现

**新增文件：`Services/Implementations/ScoreDisplayService.cs`**
```csharp
using CodeSpirit.ExamApi.Services.Interfaces;
using CodeSpirit.ExamApi.Models;
using CodeSpirit.ExamApi.Dtos.ExamPaper;
using CodeSpirit.ExamApi.Dtos.ExamRecord;
using Newtonsoft.Json.Linq;
using System.Text;

namespace CodeSpirit.ExamApi.Services.Implementations;

/// <summary>
/// 成绩显示服务实现
/// </summary>
public class ScoreDisplayService : IScoreDisplayService
{
    private readonly IScoreConversionService _scoreConversionService;
    
    public ScoreDisplayService(IScoreConversionService scoreConversionService)
    {
        _scoreConversionService = scoreConversionService;
    }
    
    /// <summary>
    /// 构建成绩显示信息
    /// </summary>
    public ScoreDisplayInfo BuildScoreDisplay(double originalScore, int originalFullScore, ExamPaperDto examPaper)
    {
        var scoreDisplay = new ScoreDisplayInfo
        {
            OriginalScore = originalScore,
            OriginalFullScore = originalFullScore,
            IsScoreConverted = examPaper.EnableScoreConversion
        };
        
        if (examPaper.EnableScoreConversion && examPaper.ConversionTargetFullScore.HasValue)
        {
            scoreDisplay.ConvertedScore = _scoreConversionService.ConvertScore(
                originalScore,
                originalFullScore,
                examPaper.ConversionTargetFullScore.Value,
                examPaper.ConversionDecimalPlaces);
            scoreDisplay.ConvertedFullScore = examPaper.ConversionTargetFullScore.Value;
            scoreDisplay.ConversionRatio = examPaper.ConversionRatio;
            scoreDisplay.ConversionDescription = _scoreConversionService.GenerateConversionDescription(
                originalFullScore,
                examPaper.ConversionTargetFullScore.Value,
                examPaper.ConversionDecimalPlaces);
            
            // 基于换算后成绩判断是否通过
            var convertedPassScore = examPaper.ConversionTargetPassScore ?? 60;
            scoreDisplay.IsPassed = scoreDisplay.ConvertedScore >= convertedPassScore;
        }
        else
        {
            // 基于原始成绩判断是否通过
            scoreDisplay.IsPassed = originalScore >= examPaper.PassScore;
        }
        
        return scoreDisplay;
    }
    
    /// <summary>
    /// 构建成绩显示信息（考试记录版本）
    /// </summary>
    public ScoreDisplayInfo BuildScoreDisplay(ExamRecordDto examRecord, ExamPaperDto examPaper)
    {
        var scoreDisplay = new ScoreDisplayInfo
        {
            OriginalScore = examRecord.OriginalScore ?? examRecord.Score ?? 0,
            OriginalFullScore = examPaper.TotalScore,
            IsScoreConverted = examRecord.IsScoreConverted,
            IsPassed = examRecord.IsPassed
        };
        
        if (examRecord.IsScoreConverted && examRecord.ConvertedScore.HasValue)
        {
            scoreDisplay.ConvertedScore = examRecord.ConvertedScore.Value;
            scoreDisplay.ConvertedFullScore = examPaper.ConversionTargetFullScore;
            scoreDisplay.ConversionRatio = examRecord.ScoreConversionRatio;
            
            if (examPaper.ConversionTargetFullScore.HasValue)
            {
                scoreDisplay.ConversionDescription = _scoreConversionService.GenerateConversionDescription(
                    examPaper.TotalScore,
                    examPaper.ConversionTargetFullScore.Value,
                    examPaper.ConversionDecimalPlaces);
            }
        }
        
        return scoreDisplay;
    }
    
    /// <summary>
    /// 生成成绩显示HTML片段
    /// </summary>
    public string GenerateScoreHtml(ScoreDisplayInfo scoreDisplay, bool showDetails = true)
    {
        var html = new StringBuilder();
        
        if (scoreDisplay.IsScoreConverted)
        {
            // 显示换算后成绩（主要显示）
            html.AppendLine($"<div class=\"score-display {scoreDisplay.ConversionCssClass}\">");
            html.AppendLine($"  <div class=\"main-score\">");
            html.AppendLine($"    <span class=\"score-value {scoreDisplay.StatusCssClass}\">{scoreDisplay.ConvertedScore:F1}</span>");
            html.AppendLine($"    <span class=\"score-unit\">/{scoreDisplay.ConvertedFullScore}分</span>");
            html.AppendLine($"    <span class=\"score-badge converted\">已换算</span>");
            html.AppendLine($"  </div>");
            
            if (showDetails)
            {
                html.AppendLine($"  <div class=\"original-score\">");
                html.AppendLine($"    <small>原始成绩：{scoreDisplay.OriginalScore:F1}/{scoreDisplay.OriginalFullScore}分</small>");
                html.AppendLine($"  </div>");
                html.AppendLine($"  <div class=\"conversion-note\">");
                html.AppendLine($"    <small class=\"text-muted\">{scoreDisplay.ConversionDescription}</small>");
                html.AppendLine($"  </div>");
            }
            
            html.AppendLine($"</div>");
        }
        else
        {
            // 显示原始成绩
            html.AppendLine($"<div class=\"score-display {scoreDisplay.ConversionCssClass}\">");
            html.AppendLine($"  <div class=\"main-score\">");
            html.AppendLine($"    <span class=\"score-value {scoreDisplay.StatusCssClass}\">{scoreDisplay.OriginalScore:F1}</span>");
            html.AppendLine($"    <span class=\"score-unit\">/{scoreDisplay.OriginalFullScore}分</span>");
            html.AppendLine($"  </div>");
            html.AppendLine($"</div>");
        }
        
        return html.ToString();
    }
    
    /// <summary>
    /// 生成成绩显示的Amis配置
    /// </summary>
    public JObject GenerateScoreAmisConfig(ScoreDisplayInfo scoreDisplay, string cssClass = "score-display")
    {
        if (scoreDisplay.IsScoreConverted)
        {
            return new JObject
            {
                ["type"] = "html",
                ["className"] = $"{cssClass} {scoreDisplay.ConversionCssClass}",
                ["html"] = $@"
                    <div class=""score-main"">
                        <div class=""score-value {scoreDisplay.StatusCssClass}"">{scoreDisplay.ConvertedScore:F1}</div>
                        <div class=""score-unit"">/ {scoreDisplay.ConvertedFullScore}分</div>
                        <div class=""score-badge converted"">已换算</div>
                    </div>
                    <div class=""score-original"">
                        <small>原始：{scoreDisplay.OriginalScore:F1}/{scoreDisplay.OriginalFullScore}分</small>
                    </div>
                    <div class=""score-note"">
                        <small class=""text-muted"">{scoreDisplay.ConversionDescription}</small>
                    </div>
                "
            };
        }
        else
        {
            return new JObject
            {
                ["type"] = "html", 
                ["className"] = $"{cssClass} {scoreDisplay.ConversionCssClass}",
                ["html"] = $@"
                    <div class=""score-main"">
                        <div class=""score-value {scoreDisplay.StatusCssClass}"">{scoreDisplay.OriginalScore:F1}</div>
                        <div class=""score-unit"">/ {scoreDisplay.OriginalFullScore}分</div>
                    </div>
                "
            };
        }
    }
    
    /// <summary>
    /// 生成成绩对比HTML（原始vs换算）
    /// </summary>
    public string GenerateScoreComparisonHtml(ScoreDisplayInfo scoreDisplay)
    {
        if (!scoreDisplay.IsScoreConverted)
        {
            return $"<div class=\"score-single\">成绩：{scoreDisplay.OriginalScore:F1}/{scoreDisplay.OriginalFullScore}分</div>";
        }
        
        var html = new StringBuilder();
        html.AppendLine("<div class=\"score-comparison\">");
        html.AppendLine("  <div class=\"comparison-row\">");
        html.AppendLine($"    <div class=\"original-score\">原始成绩：<strong>{scoreDisplay.OriginalScore:F1}/{scoreDisplay.OriginalFullScore}分</strong></div>");
        html.AppendLine("    <div class=\"conversion-arrow\">→</div>");
        html.AppendLine($"    <div class=\"converted-score\">最终成绩：<strong style=\"color: #52c41a\">{scoreDisplay.ConvertedScore:F1}/{scoreDisplay.ConvertedFullScore}分</strong></div>");
        html.AppendLine("  </div>");
        html.AppendLine($"  <div class=\"conversion-desc\"><small>{scoreDisplay.ConversionDescription}</small></div>");
        html.AppendLine("</div>");
        
        return html.ToString();
    }
}
```

### 3.1 试卷预览配置增强

**修改文件：`Controllers/ExamPapersController.cs` 中的 `GetExamQuestionsPreviewConfig` 方法：**

```csharp
[HttpGet("{id}/questions-preview")]
[DisplayName("获取试卷题目预览配置")]
public async Task<ActionResult<ApiResponse<JObject>>> GetExamQuestionsPreviewConfig(long id)
{
    var examPaper = await _examPaperService.GetAsync(id);
    if (examPaper == null)
    {
        return NotFound("试卷不存在");
    }

    // 构建成绩显示信息
    var scoreDisplay = _scoreDisplayService.BuildScoreDisplay(examPaper.TotalScore, examPaper.TotalScore, examPaper);

    // 使用JObject/JArray构建表单
    var formItems = new JArray();

    // 添加试卷信息头部，包含成绩换算提示
    var headerInfo = new JObject
    {
        ["type"] = "panel",
        ["className"] = "exam-paper-header",
        ["body"] = new JArray()
    };

    var headerBody = (JArray)headerInfo["body"];

    // 基本信息
    headerBody.Add(new JObject
    {
        ["type"] = "html",
        ["html"] = $@"
            <div class=""exam-header-info"">
                <h3>{examPaper.Name}</h3>
                <div class=""exam-basic-info"">
                    <span>总分：{examPaper.TotalScore}分</span> | 
                    <span>及格分：{examPaper.PassScore}分</span> | 
                    <span>时长：{examPaper.Duration}分钟</span> |
                    <span>题目数：{examPaper.Questions?.Count ?? 0}题</span>
                </div>
            </div>
        "
    });

    // 成绩换算提醒
    if (examPaper.EnableScoreConversion)
    {
        headerBody.Add(new JObject
        {
            ["type"] = "alert",
            ["level"] = "info",
            ["className"] = "score-conversion-alert",
            ["body"] = $@"
                <div class=""conversion-reminder"">
                    <strong>⚠️ 成绩换算提醒</strong><br/>
                    本试卷启用了成绩换算功能：{scoreDisplay.ConversionDescription}<br/>
                    考生最终成绩将按照换算规则自动转换为<strong>{examPaper.ConversionTargetFullScore}分制</strong>显示。
                    <div class=""conversion-preview"" style=""margin-top: 10px; padding: 8px; background: #f9f9f9; border-radius: 4px;"">
                        <strong>示例：</strong>原始得分{examPaper.TotalScore * 0.8:F0}分 → 换算后得分{(examPaper.TotalScore * 0.8 * (double)examPaper.ConversionRatio.Value):F1}分
                    </div>
                </div>
            "
        });
    }

    formItems.Add(headerInfo);

    // 为每个题目创建对应的表单组件
    if (examPaper.Questions != null)
    {
        for (int i = 0; i < examPaper.Questions.Count; i++)
        {
            var question = examPaper.Questions[i];
            question.Content = WebUtility.HtmlDecode(question.Content);
            question.Options = question.Options.Select(p => WebUtility.HtmlDecode(p)).ToList();
            question.CorrectAnswer = WebUtility.HtmlDecode(question.CorrectAnswer);
            question.Analysis = WebUtility.HtmlDecode(question.Analysis ?? string.Empty);
            int index = i + 1;

            };

            switch (question.Type.ToString())
```
